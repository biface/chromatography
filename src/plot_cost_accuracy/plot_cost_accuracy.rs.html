1//! Cost-vs-accuracy visualisation: Euler/RK4 solve time against
2//! $R_{sf}$(Euler, RK4), per validated case (issue #55).
3//!
4//! Run with `cargo run --example validation_report` then
5//! `cargo run --bin plot_cost_accuracy --release`; output lands in
6//! `target/plots/cost_accuracy.svg`.
7//!
8//! Reads `validation_report.json` from the system temp directory — the same
9//! file `examples/validation_report.rs` writes to, and the same path
10//! (`std::env::temp_dir()`) it prints on completion.
11//!
12//! # Visual elements
13//!
14//! 1. **Grouped bars**: Euler (blue) and RK4 (red) mean solve time per case
15//! 2. **$R_{sf}$ annotation**: printed above each case group — the accuracy
16//!    "bought" by the extra RK4 cost
17//!
18//! # Cargo.toml
19//!
20//! ```toml
21//! [[bin]]
22//! name = "plot_cost_accuracy"
23//! path = "tools/plot_cost_accuracy.rs"
24//!
25//! [dependencies]
26//! plotters   = "0.3"
27//! serde_json = "1"
28//! anyhow     = "1"
29//! ```
30
31use std::fs;
32use std::path::{Path, PathBuf};
33
34use plotters::prelude::*;
35use serde_json::Value;
36
37// =================================================================================================
38// Data
39// =================================================================================================
40
41/// One case's timing and accuracy figures, read from `validation_report.json`.
42struct CaseCostAccuracy {
43    case: String,
44    euler_ms: f64,
45    rk4_ms: f64,
46    rsf: f64,
47}
48
49fn report_path() -> PathBuf {
50    std::env::temp_dir().join("validation_report.json")
51}
52
53fn read_report(path: &Path) -> anyhow::Result<Vec<CaseCostAccuracy>> {
54    let content = fs::read_to_string(path).map_err(|e| {
55        anyhow::anyhow!(
56            "cannot read {}: {e}\nRun first: cargo run --example validation_report",
57            path.display()
58        )
59    })?;
60    let root: Value = serde_json::from_str(&content)?;
61
62    let cases = root.get("cases").and_then(Value::as_array).ok_or_else(|| {
63        anyhow::anyhow!(
64            "{}: missing or malformed top-level 'cases' array",
65            path.display()
66        )
67    })?;
68
69    cases
70        .iter()
71        .map(|c| {
72            let case = c
73                .get("case")
74                .and_then(Value::as_str)
75                .ok_or_else(|| anyhow::anyhow!("case entry missing 'case' name"))?
76                .to_string();
77            let rsf = c
78                .get("rsf_euler_vs_rk4")
79                .and_then(Value::as_f64)
80                .ok_or_else(|| anyhow::anyhow!("case '{case}' missing 'rsf_euler_vs_rk4'"))?;
81            let timing = c
82                .get("solve_time_ms")
83                .ok_or_else(|| anyhow::anyhow!("case '{case}' missing 'solve_time_ms' — regenerate the report with the current validation_report.rs"))?;
84            let euler_ms = timing
85                .get("euler")
86                .and_then(Value::as_f64)
87                .ok_or_else(|| anyhow::anyhow!("case '{case}' missing 'solve_time_ms.euler'"))?;
88            let rk4_ms = timing
89                .get("rk4")
90                .and_then(Value::as_f64)
91                .ok_or_else(|| anyhow::anyhow!("case '{case}' missing 'solve_time_ms.rk4'"))?;
92            Ok(CaseCostAccuracy { case, euler_ms, rk4_ms, rsf })
93        })
94        .collect()
95}

97// =================================================================================================
98// Plot generation
99// =================================================================================================
100
101fn generate_plot(data: &[CaseCostAccuracy], output_path: &Path) -> anyhow::Result<()> {
102    if let Some(parent) = output_path.parent() {
103        fs::create_dir_all(parent)?;
104    }
105
106    let y_max = data
107        .iter()
108        .map(|d| d.euler_ms.max(d.rk4_ms))
109        .fold(0.0_f64, f64::max)
110        * 1.3;
111
112    let root = SVGBackend::new(output_path, (1100, 650)).into_drawing_area();
113    root.fill(&WHITE)?;
114
115    let mut chart = ChartBuilder::on(&root)
116        .margin(50)
117        .x_label_area_size(45)
118        .y_label_area_size(70)
119        .caption(
120            "Solve time vs Rsf(Euler, RK4) — cost of RK4's extra accuracy",
121            ("sans-serif", 18).into_font(),
122        )
123        .build_cartesian_2d(0f64..data.len() as f64, 0f64..y_max)?;
124
125    chart
126        .configure_mesh()
127        .disable_x_mesh()
128        .x_desc("")
129        .y_desc("Mean solve time (ms)")
130        .x_label_formatter(&|_| String::new())
131        .y_label_formatter(&|v| format!("{v:.0} ms"))
132        .draw()?;
133
134    let bar_half_width = 0.35;
135
136    for (i, d) in data.iter().enumerate() {
137        let x0 = i as f64;
138        let euler_center = x0 + 0.5 - 0.22;
139        let rk4_center = x0 + 0.5 + 0.22;
140
141        chart.draw_series(std::iter::once(Rectangle::new(
142            [
143                (euler_center - bar_half_width / 2.0, 0.0),
144                (euler_center + bar_half_width / 2.0, d.euler_ms),
145            ],
146            BLUE.mix(0.7).filled(),
147        )))?;
148        chart.draw_series(std::iter::once(Rectangle::new(
149            [
150                (rk4_center - bar_half_width / 2.0, 0.0),
151                (rk4_center + bar_half_width / 2.0, d.rk4_ms),
152            ],
153            RED.mix(0.7).filled(),
154        )))?;
155
156        chart.draw_series(std::iter::once(Text::new(
157            d.case.replace('_', " "),
158            (x0 + 0.5, -y_max * 0.03),
159            ("sans-serif", 13).into_font().color(&BLACK),
160        )))?;
161
162        let top = d.euler_ms.max(d.rk4_ms);
163        chart.draw_series(std::iter::once(Text::new(
164            format!("Rsf = {:.4}", d.rsf),
165            (x0 + 0.5, top + y_max * 0.05),
166            ("sans-serif", 13).into_font().color(&RGBColor(80, 80, 80)),
167        )))?;
168    }
169
170    // Legend (manual, same convention as `plot_reference_cases_timing.rs`).
171    let w = data.len() as f64;
172    chart.draw_series(std::iter::once(Rectangle::new(
173        [(0.02 * w, y_max * 0.94), (0.06 * w, y_max * 0.98)],
174        BLUE.mix(0.7).filled(),
175    )))?;
176    chart.draw_series(std::iter::once(Text::new(
177        "Euler",
178        (0.08 * w, y_max * 0.96),
179        ("sans-serif", 12).into_font(),
180    )))?;
181    chart.draw_series(std::iter::once(Rectangle::new(
182        [(0.22 * w, y_max * 0.94), (0.26 * w, y_max * 0.98)],
183        RED.mix(0.7).filled(),
184    )))?;
185    chart.draw_series(std::iter::once(Text::new(
186        "RK4",
187        (0.28 * w, y_max * 0.96),
188        ("sans-serif", 12).into_font(),
189    )))?;
190
191    root.present()?;
192    println!("Plot generated: {}", output_path.display());
193    Ok(())
194}

196// =================================================================================================
197// Entry point
198// =================================================================================================
199
200fn main() -> anyhow::Result<()> {
201    // Standalone binary variant (Variant B) — see
202    // `chrom_rs::output::visualization::fonts` for why this call is needed
203    // before any chart renders.
204    chrom_rs::output::register_fonts();
205
206    let path = report_path();
207    println!("Reading {}...", path.display());
208    let data = read_report(&path)?;
209
210    println!(
211        "\n{:<24} {:>12} {:>12} {:>10}",
212        "case", "euler (ms)", "rk4 (ms)", "Rsf"
213    );
214    println!("{:-<62}", "");
215    for d in &data {
216        println!(
217            "{:<24} {:>12.3} {:>12.3} {:>10.4}",
218            d.case, d.euler_ms, d.rk4_ms, d.rsf
219        );
220    }
221
222    let output_path = PathBuf::from("target/plots/cost_accuracy.svg");
223    println!("\nGenerating plot...");
224    generate_plot(&data, &output_path)?;
225    Ok(())
226}

232#[cfg(test)]
233mod tests {
234    use super::*;
235
236    #[test]
237    fn test_read_report_parses_expected_shape() {
238        let json = r#"{
239            "cases": [
240                {
241                    "case": "tfa_linear",
242                    "rsf_euler_vs_rk4": 0.01,
243                    "solve_time_ms": {"euler": 5.0, "rk4": 20.0, "rk4_over_euler_ratio": 4.0}
244                }
245            ]
246        }"#;
247        let path = std::env::temp_dir().join("chrom_rs_test_cost_accuracy.json");
248        fs::write(&path, json).unwrap();
249        let data = read_report(&path).unwrap();
250        fs::remove_file(&path).ok();
251
252        assert_eq!(data.len(), 1);
253        assert_eq!(data[0].case, "tfa_linear");
254        assert!((data[0].euler_ms - 5.0).abs() < 1e-9);
255        assert!((data[0].rk4_ms - 20.0).abs() < 1e-9);
256        assert!((data[0].rsf - 0.01).abs() < 1e-9);
257    }
258
259    #[test]
260    fn test_read_report_missing_solve_time_errors() {
261        let json = r#"{"cases": [{"case": "x", "rsf_euler_vs_rk4": 0.01}]}"#;
262        let path = std::env::temp_dir().join("chrom_rs_test_cost_accuracy_missing.json");
263        fs::write(&path, json).unwrap();
264        let result = read_report(&path);
265        fs::remove_file(&path).ok();
266
267        assert!(result.is_err());
268    }
