1//! End-to-end solve time visualisation for the real reference cases (issue #55).
2//!
3//! Run with `cargo bench --bench langmuir_performance -- bench_reference_cases`
4//! then `cargo run --bin plot_reference_cases_timing --release`; output lands
5//! in `target/plots/reference_cases_timing.svg`.
6//!
7//! # Visual elements
8//!
9//! 1. **Grouped bars**: Euler (blue) and RK4 (red) mean solve time per case,
10//!    with 95% CI error bars
11//! 2. **Ratio annotation**: RK4/Euler ratio printed above each case group
12//!
13//! # Cargo.toml
14//!
15//! ```toml
16//! [[bin]]
17//! name = "plot_reference_cases_timing"
18//! path = "tools/plot_reference_cases_timing.rs"
19//!
20//! [dependencies]
21//! plotters   = "0.3"
22//! serde      = { version = "1", features = ["derive"] }
23//! serde_json = "1"
24//! anyhow     = "1"
25//! ```
26
27use std::fs;
28use std::path::{Path, PathBuf};

30use plotters::prelude::*;
31use serde::Deserialize;

33// =================================================================================================
34// Constants
35// =================================================================================================
36
37/// Case names, in the order they should appear on the x-axis — must match
38/// `RefCase::name` in `benches/langmuir_performance.rs`.
39const CASES: [&str; 2] = ["ascorbic_erythorbic", "glucose_fructose_linear"];
40
41// =================================================================================================
42// Criterion JSON deserialisation — same shape as `plot_parallelism_threshold.rs`
43// =================================================================================================
44
45#[derive(Debug, Deserialize)]
46struct ConfidenceInterval {
47    lower_bound: f64,
48    upper_bound: f64,
49}
50
51#[derive(Debug, Deserialize)]
52struct Estimate {
53    confidence_interval: ConfidenceInterval,
54    point_estimate: f64,
55}
56
57#[derive(Debug, Deserialize)]
58struct Estimates {
59    mean: Estimate,
60}
61
62// =================================================================================================
63// Data
64// =================================================================================================
65
66/// One case's measured Euler and RK4 solve time (milliseconds).
67struct CaseTiming {
68    case: &'static str,
69    euler_ms: f64,
70    euler_low_ms: f64,
71    euler_high_ms: f64,
72    rk4_ms: f64,
73    rk4_low_ms: f64,
74    rk4_high_ms: f64,
75}
76
77fn read_estimates(path: &Path) -> anyhow::Result<Estimates> {
78    let content = fs::read_to_string(path).map_err(|e| {
79        anyhow::anyhow!(
80            "cannot read {}: {e}\nRun first: cargo bench --bench langmuir_performance -- bench_reference_cases",
81            path.display()
82        )
83    })?;
84    Ok(serde_json::from_str(&content)?)
85}
86
87fn collect_timings(criterion_dir: &Path) -> anyhow::Result<Vec<CaseTiming>> {
88    let group_dir = criterion_dir.join("bench_reference_cases");
89    if !group_dir.exists() {
90        anyhow::bail!(
91            "Criterion directory not found: {}\nRun first: \
92             cargo bench --bench langmuir_performance -- bench_reference_cases",
93            group_dir.display()
94        );
95    }
96
97    let mut timings = Vec::with_capacity(CASES.len());
98    for &case in &CASES {
99        let euler = read_estimates(
100            &group_dir
101                .join("euler")
102                .join(case)
103                .join("new")
104                .join("estimates.json"),
105        )?;
106        let rk4 = read_estimates(
107            &group_dir
108                .join("rk4")
109                .join(case)
110                .join("new")
111                .join("estimates.json"),
112        )?;
113
114        timings.push(CaseTiming {
115            case,
116            euler_ms: euler.mean.point_estimate / 1e6,
117            euler_low_ms: euler.mean.confidence_interval.lower_bound / 1e6,
118            euler_high_ms: euler.mean.confidence_interval.upper_bound / 1e6,
119            rk4_ms: rk4.mean.point_estimate / 1e6,
120            rk4_low_ms: rk4.mean.confidence_interval.lower_bound / 1e6,
121            rk4_high_ms: rk4.mean.confidence_interval.upper_bound / 1e6,
122        });
123    }
124
125    Ok(timings)
126}

128// =================================================================================================
129// Plot generation

132fn generate_plot(timings: &[CaseTiming], output_path: &Path) -> anyhow::Result<()> {
133    if let Some(parent) = output_path.parent() {
134        fs::create_dir_all(parent)?;
135    }
136
137    let y_max = timings
138        .iter()
139        .map(|t| t.euler_high_ms.max(t.rk4_high_ms))
140        .fold(0.0_f64, f64::max)
141        * 1.25;
142
143    let root = SVGBackend::new(output_path, (1000, 650)).into_drawing_area();
144    root.fill(&WHITE)?;
145
146    let mut chart = ChartBuilder::on(&root)
147        .margin(50)
148        .x_label_area_size(45)
149        .y_label_area_size(70)
150        .caption(
151            "Reference cases — Euler vs RK4 end-to-end solve time",
152            ("sans-serif", 18).into_font(),
153        )
154        .build_cartesian_2d(0f64..timings.len() as f64, 0f64..y_max)?;
155
156    chart
157        .configure_mesh()
158        .disable_x_mesh()
159        .x_desc("")
160        .y_desc("Mean solve time (ms)")
161        .x_label_formatter(&|_| String::new())
162        .y_label_formatter(&|v| format!("{v:.0} ms"))
163        .draw()?;
164
165    // Bar width and per-case slot: each case occupies [i, i+1), Euler in the
166    // left half, RK4 in the right half, with a small gap between the two.
167    let bar_half_width = 0.35;
168
169    for (i, t) in timings.iter().enumerate() {
170        let x0 = i as f64;
171        let euler_center = x0 + 0.5 - 0.22;
172        let rk4_center = x0 + 0.5 + 0.22;
173
174        chart.draw_series(std::iter::once(Rectangle::new(
175            [
176                (euler_center - bar_half_width / 2.0, 0.0),
177                (euler_center + bar_half_width / 2.0, t.euler_ms),
178            ],
179            BLUE.mix(0.7).filled(),
180        )))?;
181        chart.draw_series(std::iter::once(PathElement::new(
182            vec![
183                (euler_center, t.euler_low_ms),
184                (euler_center, t.euler_high_ms),
185            ],
186            ShapeStyle {
187                color: BLUE.mix(0.9).to_rgba(),
188                filled: false,
189                stroke_width: 2,
190            },
191        )))?;
192
193        chart.draw_series(std::iter::once(Rectangle::new(
194            [
195                (rk4_center - bar_half_width / 2.0, 0.0),
196                (rk4_center + bar_half_width / 2.0, t.rk4_ms),
197            ],
198            RED.mix(0.7).filled(),
199        )))?;
200        chart.draw_series(std::iter::once(PathElement::new(
201            vec![(rk4_center, t.rk4_low_ms), (rk4_center, t.rk4_high_ms)],
202            ShapeStyle {
203                color: RED.mix(0.9).to_rgba(),
204                filled: false,
205                stroke_width: 2,
206            },
207        )))?;
208
209        // Case name, below the axis.
210        chart.draw_series(std::iter::once(Text::new(
211            t.case.replace('_', " "),
212            (x0 + 0.5, -y_max * 0.03),
213            ("sans-serif", 13).into_font().color(&BLACK),
214        )))?;
215
216        // Ratio annotation, above the taller bar.
217        let ratio = t.rk4_ms / t.euler_ms;
218        let top = t.euler_high_ms.max(t.rk4_high_ms);
219        chart.draw_series(std::iter::once(Text::new(
220            format!("×{ratio:.2}"),
221            (x0 + 0.5, top + y_max * 0.04),
222            ("sans-serif", 13).into_font().color(&BLACK),
223        )))?;
224    }
225
226    // Legend (manual — no data series carries a `.label()` in this layout).
227    chart.draw_series(std::iter::once(Rectangle::new(
228        [
229            (0.02 * timings.len() as f64, y_max * 0.94),
230            (0.06 * timings.len() as f64, y_max * 0.98),
231        ],
232        BLUE.mix(0.7).filled(),
233    )))?;
234    chart.draw_series(std::iter::once(Text::new(
235        "Euler",
236        (0.08 * timings.len() as f64, y_max * 0.96),
237        ("sans-serif", 12).into_font(),
238    )))?;
239    chart.draw_series(std::iter::once(Rectangle::new(
240        [
241            (0.22 * timings.len() as f64, y_max * 0.94),
242            (0.26 * timings.len() as f64, y_max * 0.98),
243        ],
244        RED.mix(0.7).filled(),
245    )))?;
246    chart.draw_series(std::iter::once(Text::new(
247        "RK4",
248        (0.28 * timings.len() as f64, y_max * 0.96),
249        ("sans-serif", 12).into_font(),
250    )))?;
251
252    root.present()?;
253    println!("Plot generated: {}", output_path.display());
254    Ok(())
255}

257// =================================================================================================
258// Entry point
259// =================================================================================================
260
261fn main() -> anyhow::Result<()> {
262    let criterion_dir = PathBuf::from("target/criterion");
263    let output_path = PathBuf::from("target/plots/reference_cases_timing.svg");
264
265    println!("Reading Criterion data...");
266    let timings = collect_timings(&criterion_dir)?;
267
268    println!(
269        "\n{:<24} {:>12} {:>12} {:>10}",
270        "case", "euler (ms)", "rk4 (ms)", "ratio"
271    );
272    println!("{:-<60}", "");
273    for t in &timings {
274        println!(
275            "{:<24} {:>12.3} {:>12.3} {:>9.2}×",
276            t.case,
277            t.euler_ms,
278            t.rk4_ms,
279            t.rk4_ms / t.euler_ms
280        );
281    }
282
283    println!("\nGenerating plot...");
284    generate_plot(&timings, &output_path)?;
285    Ok(())
286}
