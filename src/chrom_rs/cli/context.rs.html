1//! Execution context shared across all `chrom-rs` command handlers.
2//!
3//! - [`ChromContext`]: the `project_dir` invariant plus the three
4//!   pending-configuration builder slots used by `config`/`build` (DD-016).
5//! - [`ContextError`]: errors from [`ChromContext::set_project_dir`].
6//!
7//! # Path invariants
8//!
9//! `project_dir` is always a *safe* directory:
10//! - No `..` component (prevents escaping the declared root).
11//! - The directory exists and is readable by the current process.
12//! - The directory is writable by the current process (output files land here).
13//!
14//! These invariants are enforced exclusively by
15//! [`ChromContext::set_project_dir`].
16

31/// Errors that can occur when configuring a [`ChromContext`].
32#[derive(Debug)]
33pub enum ContextError {
34    /// The path contains a `..` component.
35    PathTraversal(PathBuf),
36
37    /// The path does not point to an existing directory.
38    NotADirectory(PathBuf),
39
40    /// The current process lacks read or write permission on the directory.
41    PermissionDenied(PathBuf, io::Error),
42}
43
44impl std::fmt::Display for ContextError {
45    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
46        match self {
47            ContextError::PathTraversal(p) => write!(
48                f,
49                "project-dir '{}' contains '..': path traversal is not allowed",
50                p.display()
51            ),
52            ContextError::NotADirectory(p) => write!(
53                f,
54                "project-dir '{}' does not exist or is not a directory",
55                p.display()
56            ),
57            ContextError::PermissionDenied(p, e) => write!(
58                f,
59                "project-dir '{}': insufficient permissions — {e}",
60                p.display()
61            ),
62        }
63    }
64}
65
66impl std::error::Error for ContextError {
67    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
68        match self {
69            ContextError::PermissionDenied(_, e) => Some(e),
70            _ => None,
71        }
72    }
73}

79/// Runtime state shared across all `chrom-rs` command handlers.
80///
81/// `project_dir` is the directory relative to which `--model`,
82/// `--scenario`, `--solver`, and all output file names are resolved.
83///
84/// Since v0.6.0, `ChromContext` also holds three optional builder slots —
85/// `pending_model`, `pending_solver`, `pending_scenario` — one per
86/// `config`/`build` target (DD-016, issue #53). Each slot is filled in
87/// field by field across one or more `config` occurrences, whether given in
88/// a single invocation or spread across a chained sequence (`dynamic-cli`
89/// 0.9.0 command chaining). Nothing in a slot is validated until it is
90/// turned into a real model/solver/scenario by `save` or `run`.
91///
92/// # Example
93///
94/// ```rust,no_run
95/// use chrom_rs::cli::context::ChromContext;
96///
97/// let mut ctx = ChromContext::new();           // project_dir = "."
98/// ctx.set_project_dir("experiments/run_01").unwrap();
99/// assert_eq!(ctx.project_dir(), std::path::Path::new("experiments/run_01"));
100/// ```
101pub struct ChromContext {
102    /// Root directory for all file-name resolution.
103    ///
104    /// Invariants enforced by [`set_project_dir`](Self::set_project_dir):
105    /// no `..` component; existing, readable, writable directory.
106    project_dir: PathBuf,
107
108    /// Pending model under construction via `config --model ...`.
109    /// `None` until the first `single`/`multi`/`species` occurrence.
110    pending_model: Option<ModelBuilder>,
111
112    /// Pending solver under construction via `config --solver ...`.
113    pending_solver: Option<SolverBuilder>,
114
115    /// Pending scenario under construction via `config --scenario ...`.
116    pending_scenario: Option<ScenarioBuilder>,
117}
118
119impl ChromContext {
120    /// Creates a context whose project directory is the current working
121    /// directory (`.`).
122    pub fn new() -> Self {
123        Self {
124            project_dir: PathBuf::from("."),
125            pending_model: None,
126            pending_solver: None,
127            pending_scenario: None,
128        }
129    }
130
131    /// Returns the current project directory.
132    pub fn project_dir(&self) -> &Path {
133        &self.project_dir
134    }
135
136    /// Returns the pending model slot, if any.
137    pub fn pending_model(&self) -> Option<&ModelBuilder> {
138        self.pending_model.as_ref()
139    }
140
141    /// Returns the pending solver slot, if any.
142    pub fn pending_solver(&self) -> Option<&SolverBuilder> {
143        self.pending_solver.as_ref()
144    }
145
146    /// Returns the pending scenario slot, if any.
147    pub fn pending_scenario(&self) -> Option<&ScenarioBuilder> {
148        self.pending_scenario.as_ref()
149    }
150
151    /// Merges `fields` into the pending model as a single-species build.
152    ///
153    /// - If the slot is empty, locks it to `Single(fields)`.
154    /// - If the slot already holds `Single`, merges field by field
155    ///   (last-write-wins per field, see [`SingleModelBuilder::merge`]).
156    /// - If the slot holds `Multi`, resets it to `Single(fields)` and
157    ///   returns [`ShapeSwitch::ToSingle`] so the caller can print a
158    ///   visible warning — the previously accumulated multi-species state
159    ///   (including any species list) is discarded, not merged.
160    pub fn merge_model_single(&mut self, fields: SingleModelBuilder) -> Option<ShapeSwitch> {
161        match &mut self.pending_model {
162            None => {
163                self.pending_model = Some(ModelBuilder::Single(fields));
164                None
165            }
166            Some(ModelBuilder::Single(existing)) => {
167                existing.merge(fields);
168                None
169            }
170            Some(ModelBuilder::Multi(_)) => {
171                self.pending_model = Some(ModelBuilder::Single(fields));
172                Some(ShapeSwitch::ToSingle)
173            }
174        }
175    }
176
177    /// Merges the scalar fields of `fields` into the pending model as a
178    /// multi-species build. Mirrors [`Self::merge_model_single`]'s
179    /// switch/merge/lock logic, in the opposite direction. `fields.species`
180    /// is ignored here — add species via [`Self::add_species`] instead.
181    pub fn merge_model_multi(&mut self, fields: MultiModelBuilder) -> Option<ShapeSwitch> {
182        match &mut self.pending_model {
183            None => {
184                self.pending_model = Some(ModelBuilder::Multi(fields));
185                None
186            }
187            Some(ModelBuilder::Multi(existing)) => {
188                existing.merge_scalars(fields);
189                None
190            }
191            Some(ModelBuilder::Single(_)) => {
192                self.pending_model = Some(ModelBuilder::Multi(fields));
193                Some(ShapeSwitch::ToMulti)
194            }
195        }
196    }
197
198    /// Appends one species to the pending multi-species model, in call
199    /// order. A `species` occurrence alone is enough to lock the slot into
200    /// `Multi` — a scalar `multi` occurrence is not required first.
201    ///
202    /// If the slot currently holds `Single`, it is reset to an empty
203    /// `Multi` (species list starting with just this one) and
204    /// [`ShapeSwitch::ToMulti`] is returned as a visible-warning signal.
205    pub fn add_species(&mut self, species: SpeciesBuilder) -> Option<ShapeSwitch> {
206        match &mut self.pending_model {
207            None => {
208                let mut multi = MultiModelBuilder::default();
209                multi.push_species(species);
210                self.pending_model = Some(ModelBuilder::Multi(multi));
211                None
212            }
213            Some(ModelBuilder::Multi(existing)) => {
214                existing.push_species(species);
215                None
216            }
217            Some(ModelBuilder::Single(_)) => {
218                let mut multi = MultiModelBuilder::default();
219                multi.push_species(species);
220                self.pending_model = Some(ModelBuilder::Multi(multi));
221                Some(ShapeSwitch::ToMulti)
222            }
223        }
224    }
225
226    /// Merges `fields` into the pending solver, field by field
227    /// (last-write-wins per field). Locks the slot on first call.
228    pub fn merge_solver(&mut self, fields: SolverBuilder) {
229        match &mut self.pending_solver {
230            Some(existing) => existing.merge(fields),
231            None => self.pending_solver = Some(fields),
232        }
233    }
234
235    /// Sets or merges the pending scenario's initial condition.
236    pub fn set_scenario_initial_condition(&mut self, value: impl Into<String>) {
237        self.pending_scenario
238            .get_or_insert_with(ScenarioBuilder::default)
239            .set_initial_condition(value.into());
240    }
241
242    /// Merges `injection` into the pending scenario's default injection.
243    pub fn merge_scenario_default_injection(&mut self, injection: InjectionBuilder) {
244        self.pending_scenario
245            .get_or_insert_with(ScenarioBuilder::default)
246            .merge_default_injection(injection);
247    }
248
249    /// Merges `injection` into the pending scenario's override for
250    /// `species` (creating the override on first mention of that species).
251    pub fn merge_scenario_species_override(&mut self, species: &str, injection: InjectionBuilder) {
252        self.pending_scenario
253            .get_or_insert_with(ScenarioBuilder::default)
254            .merge_species_override(species, injection);
255    }
256
257    /// Sets the project directory after validating the path.
258    ///
259    /// # Validation
260    ///
261    /// 1. Rejects any path containing a `..` component.
262    /// 2. Checks that the path points to an existing directory.
263    /// 3. Verifies read permission by listing the directory.
264    /// 4. Verifies write permission by creating and removing a probe file.
265    ///
266    /// # Errors
267    ///
268    /// - [`ContextError::PathTraversal`] if `path` contains `..`.
269    /// - [`ContextError::NotADirectory`] if `path` does not exist or is a file.
270    /// - [`ContextError::PermissionDenied`] if the process cannot read or
271    ///   write the directory.
272    pub fn set_project_dir(&mut self, path: impl Into<PathBuf>) -> Result<(), ContextError> {
273        let path = path.into();
274
275        // 1 — Reject `..` regardless of position.
276        if path.components().any(|c| c == Component::ParentDir) {
277            return Err(ContextError::PathTraversal(path));
278        }
279
280        // 2 — Must be an existing directory.
281        if !path.is_dir() {
282            return Err(ContextError::NotADirectory(path));
283        }
284
285        // 3 — Read permission.
286        std::fs::read_dir(&path).map_err(|e| ContextError::PermissionDenied(path.clone(), e))?;
287
288        // 4 — Write permission: probe with a temporary file.
289        let probe = path.join(".chrom_rs_write_probe");
290        std::fs::File::create(&probe)
291            .map_err(|e| ContextError::PermissionDenied(path.clone(), e))?;
292        let _ = std::fs::remove_file(&probe);
293
294        self.project_dir = path;
295        Ok(())
296    }
297}
