1//! `check` command — validates configuration files without running a
2//! simulation.
3//!
4//! Two independent things, both requested explicitly, neither guessed at:
5//!
6//! 1. **Directory inventory.** Lists every `.yml`/`.yaml`/`.json` file
7//!    directly under `--project-dir`, with a warning if fewer than three are
8//!    present. No attempt is made to classify a file as "the" model,
9//!    scenario, or solver — a directory can legitimately hold several
10//!    variants of each (see `examples/config/tfa/`), and guessing which one
11//!    the user means would be worse than not guessing at all.
12//! 2. **Targeted validation**, for whichever `--source <role> file=...`
13//!    occurrences are actually given (zero, one, two, or three — unlike
14//!    `run`, no role is required here). Each is loaded with the same
15//!    `load_model`/`load_scenario`/`load_solver` functions `run` itself
16//!    uses, so "check passes" and "run would accept this file" mean exactly
17//!    the same thing — there is no separate, weaker validation path to fall
18//!    out of sync with the real one. `scenario` can only be validated
19//!    alongside `model` (`load_scenario` needs a live model to check
20//!    per-species injections against — see its doc comment), so a
21//!    `scenario` given without a `model` is reported as skipped, not
22//!    silently ignored and not treated as a failure on its own.
23//!
24//! `check` has no legacy scalar equivalents (`--model`, `--output-csv`,
25//! …) — it's a new command, introduced alongside the new `--source`
26//! syntax (both build on `dynamic-cli`'s own repeatable-sub-parameter
27//! feature, [dcli#21](https://github.com/biface/dcli/issues/21)), so
28//! there's no prior syntax to stay compatible with.
29

41/// `check` command handler.
42///
43/// See the module-level doc comment for what it does and, as importantly,
44/// what it deliberately does not do (no cross-combination testing, no file
45/// classification by naming convention or trial parsing).
46pub struct CheckHandler;
47
48impl CommandHandler for CheckHandler {
49    fn execute(
50        &self,
51        ctx: &mut dyn ExecutionContext,
52        args: &ParsedArgs,
53    ) -> dynamic_cli::Result<()> {
54        // ── 1. Project directory ─────────────────────────────────────────────
55        let chrom_ctx = ctx
56            .as_any_mut()
57            .downcast_mut::<ChromContext>()
58            .ok_or_else(|| {
59                DynamicCliError::from(ExecutionError::ContextDowncastFailed {
60                    expected_type: "ChromContext".to_string(),
61                    suggestion: None,
62                })
63            })?;
64
65        let project_dir_str = args.get_scalar("project-dir").unwrap_or(".");
66        chrom_ctx
67            .set_project_dir(project_dir_str)
68            .map_err(|e| to_cli_err(anyhow!("{e}")))?;
69        let project_dir: PathBuf = chrom_ctx.project_dir().to_path_buf();
70
71        // ── 2. Directory inventory ───────────────────────────────────────────
72        println!("Project directory: {}", project_dir.display());
73
74        let mut config_like: Vec<PathBuf> = fs::read_dir(&project_dir)
75            .map_err(|e| to_cli_err(anyhow!("reading project directory: {e}")))?
76            .filter_map(|entry| entry.ok())
77            .map(|entry| entry.path())
78            .filter(|path| {
79                path.extension()
80                    .and_then(|ext| ext.to_str())
81                    .is_some_and(|ext| matches!(ext, "yml" | "yaml" | "json"))
82            })
83            .collect();
84        config_like.sort();
85
86        println!("Config-like files found: {}", config_like.len());
87        for f in &config_like {
88            let name = f
89                .file_name()
90                .map(|n| n.to_string_lossy().into_owned())
91                .unwrap_or_default();
92            println!("  {name}");
93        }
94        if config_like.len() < 3 {
95            println!(
96                "Note: fewer than 3 config-like files present — a typical project needs \
97                 at least a model, a scenario, and a solver config."
98            );
99        }
100        println!();
101
102        // ── 3. Targeted validation ───────────────────────────────────────────
103        let model_file = resolve_source_optional(args, "model").map_err(to_cli_err)?;
104        let scenario_file = resolve_source_optional(args, "scenario").map_err(to_cli_err)?;
105        let solver_file = resolve_source_optional(args, "solver").map_err(to_cli_err)?;
106
107        let mut checked_anything = false;
108        let mut any_failed = false;
109
110        let loaded_model = match &model_file {
111            Some(file) => {
112                checked_anything = true;
113                let path = project_dir.join(file);
114                let path_str = path_to_str(&path).map_err(to_cli_err)?;
115                match load_model(path_str) {
116                    Ok(model) => {
117                        println!("model    OK    {file}");
118                        Some(model)
119                    }
120                    Err(e) => {
121                        any_failed = true;
122                        println!("model    FAIL  {file} — {e}");
123                        None
124                    }
125                }
126            }
127            None => None,
128        };
129
130        if let Some(file) = &scenario_file {
131            checked_anything = true;
132            match loaded_model {
133                Some(mut model) => {
134                    let path = project_dir.join(file);
135                    let path_str = path_to_str(&path).map_err(to_cli_err)?;
136                    match load_scenario(path_str, &mut *model) {
137                        Ok(_) => println!("scenario OK    {file}"),
138                        Err(e) => {
139                            any_failed = true;
140                            println!("scenario FAIL  {file} — {e}");
141                        }
142                    }
143                }
144                None if model_file.is_some() => {
145                    // A model was given but failed to load — already
146                    // reported above; scenario can't be checked against it.
147                    println!("scenario SKIP  {file} — model failed to load, see above");
148                }
149                None => {
150                    println!(
151                        "scenario SKIP  {file} — cannot validate without \
152                         '--source model file=...' (injections are checked against the model)"
153                    );
154                }
155            }
156        }
157
158        if let Some(file) = &solver_file {
159            checked_anything = true;
160            let path = project_dir.join(file);
161            let path_str = path_to_str(&path).map_err(to_cli_err)?;
162            match load_solver(path_str) {
163                Ok(_) => println!("solver   OK    {file}"),
164                Err(e) => {
165                    any_failed = true;
166                    println!("solver   FAIL  {file} — {e}");
167                }
168            }
169        }
170
171        if !checked_anything {
172            println!("No '--source' given — directory inventory only, nothing validated.");
173        }
174
175        if any_failed {
176            return Err(to_cli_err(anyhow!(
177                "one or more configuration files failed validation"
178            )));
179        }
180
181        Ok(())
182    }
183}

196    /// Builds a `ParsedArgs` with a "project-dir" scalar plus a "source" key
197    /// carrying the given (discriminant, file) occurrences.
198    fn args_with_source(project_dir: &std::path::Path, occurrences: &[(&str, &str)]) -> ParsedArgs {
199        let occs: Vec<OptionOccurrence> = occurrences
200            .iter()
201            .map(|(discriminant, file)| OptionOccurrence {
202                discriminant: discriminant.to_string(),
203                params: HashMap::from([("file".to_string(), file.to_string())]),
204            })
205            .collect();
206        let mut map = HashMap::new();
207        map.insert(
208            "project-dir".to_string(),
209            ParsedValue::Scalar(project_dir.to_string_lossy().into_owned()),
210        );
211        map.insert("source".to_string(), ParsedValue::Repeated(occs));
212        ParsedArgs::new(map)
213    }
214
215    fn temp_project_dir() -> tempfile::TempDir {
216        tempfile::tempdir().expect("failed to create temp dir")
217    }
218
219    fn write_file(dir: &std::path::Path, name: &str, content: &str) -> PathBuf {
220        let path = dir.join(name);
221        let mut f = std::fs::File::create(&path).expect("failed to create file");
222        f.write_all(content.as_bytes())
223            .expect("failed to write file");
224        path
225    }
226
227    #[test]
228    fn test_check_no_source_lists_directory_only() {
229        let dir = temp_project_dir();
230        write_file(dir.path(), "model.yml", "LangmuirSingle:\n  lambda: 1.0\n");
231        write_file(dir.path(), "notes.txt", "not a config file");
232
233        let mut ctx = ChromContext::new();
234        let args = ParsedArgs::from_scalars(HashMap::from([(
235            "project-dir".to_string(),
236            dir.path().to_string_lossy().into_owned(),
237        )]));
238
239        let result = CheckHandler.execute(&mut ctx, &args);
240        assert!(result.is_ok(), "check with no --source should never fail");
241    }
242
243    #[test]
244    fn test_check_scenario_without_model_is_skipped_not_failed() {
245        let dir = temp_project_dir();
246        write_file(
247            dir.path(),
248            "scenario.yml",
249            "initial_condition: zero\ndefault_injection:\n  type: None\n",
250        );
251
252        let mut ctx = ChromContext::new();
253        let args = args_with_source(dir.path(), &[("scenario", "scenario.yml")]);
254
255        let result = CheckHandler.execute(&mut ctx, &args);
256        assert!(
257            result.is_ok(),
258            "scenario-without-model should be a skip, not a failure"
259        );
260    }
261
262    #[test]
263    fn test_check_missing_source_file_fails() {
264        let dir = temp_project_dir();
265
266        let mut ctx = ChromContext::new();
267        let args = args_with_source(dir.path(), &[("model", "does-not-exist.yml")]);
268
269        let result = CheckHandler.execute(&mut ctx, &args);
270        assert!(
271            result.is_err(),
272            "a --source pointing at a nonexistent file must fail check"
273        );
274    }
