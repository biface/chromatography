1//! `save` command — serialise a pending `config` builder slot to a real
2//! `model.yml` / `solver.yml` / `scenario.yml` file, in exactly the shape
3//! [`crate::config::model::load_model`],
4//! [`crate::config::solver::load_solver`], and
5//! [`crate::config::scenario::load_scenario`] already expect to read back
6//! (DD-016, [#53](https://github.com/biface/chromatography/issues/53),
7//! [#72](https://github.com/biface/chromatography/issues/72)).
8//!
9//! `LangmuirSingle`/`LangmuirMulti`/`SpeciesParams`'s own fields are
10//! private, and their public constructors take raw physical inputs
11//! (`porosity`, `velocity`) rather than the already-precomputed
12//! `dz`/`fe`/`ue` a pending model carries — so this module builds the YAML
13//! directly as a [`serde_yaml::Value`] tree instead of instantiating the
14//! real structs, matching field names and the `LangmuirSingle:`/
15//! `LangmuirMulti:` typetag root key by hand.
16//!
17//! Every target validates its own required fields before writing anything:
18//! a missing field is an explicit, named error, never a YAML file with a
19//! `null` in place of a value `load_model`/`load_solver` would then reject
20//! with a much less helpful message.
21

39impl CommandHandler for SaveHandler {
40    fn execute(
41        &self,
42        ctx: &mut dyn ExecutionContext,
43        args: &ParsedArgs,
44    ) -> dynamic_cli::Result<()> {
45        let chrom_ctx = ctx
46            .as_any_mut()
47            .downcast_mut::<ChromContext>()
48            .ok_or_else(|| {
49                DynamicCliError::from(ExecutionError::ContextDowncastFailed {
50                    expected_type: "ChromContext".to_string(),
51                    suggestion: None,
52                })
53            })?;
54
55        let project_dir_str = args.get_scalar("project-dir").unwrap_or(".");
56        chrom_ctx
57            .set_project_dir(project_dir_str)
58            .map_err(|e| to_cli_err(anyhow!("{e}")))?;
59        let project_dir: PathBuf = chrom_ctx.project_dir().to_path_buf();
60
61        let target = args
62            .get_scalar("target")
63            .ok_or_else(|| to_cli_err(anyhow!("--target is required")))?;
64        let file = args
65            .get_scalar("file")
66            .ok_or_else(|| to_cli_err(anyhow!("--file is required")))?;
67
68        let content = match target {
69            "model-single" => match chrom_ctx.pending_model() {
70                Some(ModelBuilder::Single(single)) => {
71                    build_model_single(single).map_err(to_cli_err)?
72                }
73                Some(ModelBuilder::Multi(_)) => {
74                    return Err(to_cli_err(anyhow!(
75                        "cannot save 'model-single': the pending model is currently \
76                         multi-species — use --target model-multi, or start over with \
77                         config --model single ..."
78                    )));
79                }
80                None => {
81                    return Err(to_cli_err(anyhow!(
82                        "cannot save 'model-single': no pending model in this session \
83                         (config --model single ... first)"
84                    )));
85                }
86            },
87            "model-multi" => match chrom_ctx.pending_model() {
88                Some(ModelBuilder::Multi(multi)) => build_model_multi(multi).map_err(to_cli_err)?,
89                Some(ModelBuilder::Single(_)) => {
90                    return Err(to_cli_err(anyhow!(
91                        "cannot save 'model-multi': the pending model is currently \
92                         single-species — use --target model-single, or start over with \
93                         config --model multi/species ..."
94                    )));
95                }
96                None => {
97                    return Err(to_cli_err(anyhow!(
98                        "cannot save 'model-multi': no pending model in this session \
99                         (config --model multi/species ... first)"
100                    )));
101                }
102            },
103            "solver" => {
104                let solver = chrom_ctx.pending_solver().ok_or_else(|| {
105                    to_cli_err(anyhow!(
106                        "cannot save 'solver': no pending solver in this session \
107                         (config --solver ... first)"
108                    ))
109                })?;
110                build_solver(solver).map_err(to_cli_err)?
111            }
112            "scenario" => {
113                // Unlike model-single/model-multi/solver, an entirely empty
114                // scenario is a legitimate save: every scenario.yml field is
115                // optional to load_scenario (initial_condition defaults to
116                // "zero", default_injection/injections default to none) —
117                // there is no required-field error path for this target.
118                let scenario = chrom_ctx.pending_scenario().cloned().unwrap_or_default();
119                build_scenario(&scenario).map_err(to_cli_err)?
120            }
121            other => {
122                // Defensive only: commands.yml's `choices` list means dcli
123                // itself rejects anything else before this handler runs.
124                return Err(to_cli_err(anyhow!("unsupported --target '{other}'")));
125            }
126        };
127
128        let file_path = project_dir.join(file);
129        std::fs::write(&file_path, content).map_err(|e| {
130            to_cli_err(anyhow!(
131                "failed to write '{}': {e}",
132                path_to_str(&file_path).unwrap_or(file)
133            ))
134        })?;
135
136        Ok(())
137    }
138}

144/// Returns `value`, or a named "required field missing" error mentioning
145/// both the field and the target — never lets a `None` slip through to
146/// become a YAML `null`.
147fn require<T>(value: Option<T>, field: &str, target: &str) -> anyhow::Result<T> {
148    value.ok_or_else(|| anyhow!("cannot save '{target}': required field '{field}' is not set"))
149}
150
151/// `{ type: None }` — the placeholder every `model.yml` injection field
152/// carries; real injections are supplied by `scenario.yml` and applied by
153/// `scenario::load_scenario` after the model loads, never by `model.yml`
154/// itself.
155fn none_injection() -> Value {
156    let mut m = Mapping::new();
157    m.insert("type".into(), "None".into());
158    Value::Mapping(m)
159}

161/// Builds `model.yml`'s `LangmuirSingle:`-rooted shape from a
162/// [`SingleModelBuilder`]. All eight fields are required — see the module
163/// doc comment for why this doesn't go through `LangmuirSingle` itself.
164fn build_model_single(single: &SingleModelBuilder) -> anyhow::Result<String> {
165    const TARGET: &str = "model-single";
166    let lambda = require(single.lambda, "lambda", TARGET)?;
167    let langmuir_k = require(single.langmuir_k, "langmuir-k", TARGET)?;
168    let port_number = require(single.port_number, "port-number", TARGET)?;
169    let column_length = require(single.column_length, "column-length", TARGET)?;
170    let n_points = require(single.n_points, "n-points", TARGET)?;
171    let dz = require(single.dz, "dz", TARGET)?;
172    let fe = require(single.fe, "fe", TARGET)?;
173    let ue = require(single.ue, "ue", TARGET)?;
174
175    let mut fields = Mapping::new();
176    fields.insert("lambda".into(), lambda.into());
177    fields.insert("langmuir_k".into(), langmuir_k.into());
178    fields.insert("port_number".into(), port_number.into());
179    fields.insert("column_length".into(), column_length.into());
180    fields.insert("n_points".into(), n_points.into());
181    fields.insert("dz".into(), dz.into());
182    fields.insert("fe".into(), fe.into());
183    fields.insert("ue".into(), ue.into());
184    fields.insert("injection".into(), none_injection());
185
186    let mut root = Mapping::new();
187    root.insert("LangmuirSingle".into(), Value::Mapping(fields));
188
189    serde_yaml::to_string(&Value::Mapping(root))
190        .map_err(|e| anyhow!("failed to serialise {TARGET}: {e}"))
191}

193/// Builds `model.yml`'s `LangmuirMulti:`-rooted shape from a
194/// [`MultiModelBuilder`]. All eight scalar fields are required, at least
195/// one species is required, and every species must itself have all four
196/// fields set — a species missing `name` should be unreachable in practice
197/// (`commands.yml` marks it `required: true`), but is still checked here
198/// defensively rather than assumed.
199fn build_model_multi(multi: &MultiModelBuilder) -> anyhow::Result<String> {
200    const TARGET: &str = "model-multi";
201    let n_points = require(multi.n_points, "n-points", TARGET)?;
202    let porosity = require(multi.porosity, "porosity", TARGET)?;
203    let velocity = require(multi.velocity, "velocity", TARGET)?;
204    let column_length = require(multi.column_length, "column-length", TARGET)?;
205    let dz = require(multi.dz, "dz", TARGET)?;
206    let fe = require(multi.fe, "fe", TARGET)?;
207    let ue = require(multi.ue, "ue", TARGET)?;
208    let stationary_fraction = require(multi.stationary_fraction, "stationary-fraction", TARGET)?;
209
210    if multi.species.is_empty() {
211        return Err(anyhow!(
212            "cannot save '{TARGET}': at least one species is required \
213             (config --model species name=... ...)"
214        ));
215    }
216
217    let mut species_seq = Vec::with_capacity(multi.species.len());
218    for (index, species) in multi.species.iter().enumerate() {
219        let unnamed_context = format!("{TARGET} (species #{index})");
220        let name = require(species.name.clone(), "name", &unnamed_context)?;
221        let named_context = format!("{TARGET} species '{name}'");
222        let lambda = require(species.lambda, "lambda", &named_context)?;
223        let langmuir_k = require(species.langmuir_k, "langmuir-k", &named_context)?;
224        let port_number = require(species.port_number, "port-number", &named_context)?;
225
226        let mut sp = Mapping::new();
227        sp.insert("name".into(), name.into());
228        sp.insert("lambda".into(), lambda.into());
229        sp.insert("langmuir_k".into(), langmuir_k.into());
230        sp.insert("port_number".into(), port_number.into());
231        sp.insert("injection".into(), none_injection());
232        species_seq.push(Value::Mapping(sp));
233    }
234
235    let mut fields = Mapping::new();
236    fields.insert("species".into(), Value::Sequence(species_seq));
237    fields.insert("n_points".into(), n_points.into());
238    fields.insert("porosity".into(), porosity.into());
239    fields.insert("velocity".into(), velocity.into());
240    fields.insert("column_length".into(), column_length.into());
241    fields.insert("dz".into(), dz.into());
242    fields.insert("fe".into(), fe.into());
243    fields.insert("ue".into(), ue.into());
244    fields.insert("stationary_fraction".into(), stationary_fraction.into());
245
246    let mut root = Mapping::new();
247    root.insert("LangmuirMulti".into(), Value::Mapping(fields));
248
249    serde_yaml::to_string(&Value::Mapping(root))
250        .map_err(|e| anyhow!("failed to serialise {TARGET}: {e}"))
251}

253/// Builds `solver.yml`'s flat (no root wrapper key) shape from a
254/// [`SolverBuilder`]. `type`/`total_time`/`time_steps` are required; `step`
255/// is written only when set — an absent key means the same "full
256/// trajectory" default as an explicit `null` to `load_solver`, so omitting
257/// it is preferred over writing a literal `null`.
258fn build_solver(solver: &SolverBuilder) -> anyhow::Result<String> {
259    const TARGET: &str = "solver";
260    let solver_type = require(solver.solver_type.clone(), "type", TARGET)?;
261    let total_time = require(solver.total_time, "total-time", TARGET)?;
262    let time_steps = require(solver.time_steps, "time-steps", TARGET)?;
263
264    let mut fields = Mapping::new();
265    fields.insert("type".into(), solver_type.into());
266    fields.insert("total_time".into(), total_time.into());
267    fields.insert("time_steps".into(), time_steps.into());
268    if let Some(step) = solver.step {
269        fields.insert("step".into(), step.into());
270    }
271
272    serde_yaml::to_string(&Value::Mapping(fields))
273        .map_err(|e| anyhow!("failed to serialise {TARGET}: {e}"))
274}

276/// Builds `scenario.yml`'s flat shape from a [`ScenarioBuilder`]. Every
277/// field is optional — see the `"scenario"` match arm in
278/// [`SaveHandler::execute`] for why there is no required-field error path
279/// for this target.
280fn build_scenario(scenario: &ScenarioBuilder) -> anyhow::Result<String> {
281    let mut fields = Mapping::new();
282
283    if let Some(initial_condition) = &scenario.initial_condition {
284        fields.insert("initial_condition".into(), initial_condition.clone().into());
285    }
286
287    if let Some(default) = &scenario.default_injection {
288        fields.insert(
289            "default_injection".into(),
290            injection_to_value(default, "scenario default_injection")?,
291        );
292    }
293
294    if !scenario.species_overrides.is_empty() {
295        let mut overrides = Vec::with_capacity(scenario.species_overrides.len());
296        for (species, injection) in &scenario.species_overrides {
297            let context = format!("scenario injection override for '{species}'");
298            let mut entry = match injection_to_value(injection, &context)? {
299                Value::Mapping(m) => m,
300                _ => unreachable!("injection_to_value always returns a Mapping"),
301            };
302            entry.insert("species".into(), species.clone().into());
303            overrides.push(Value::Mapping(entry));
304        }
305        fields.insert("injections".into(), Value::Sequence(overrides));
306    }
307
308    serde_yaml::to_string(&Value::Mapping(fields))
309        .map_err(|e| anyhow!("failed to serialise scenario: {e}"))
310}

312/// Builds one injection block (`default_injection` or one entry of
313/// `injections`). `type` is required and selects which further fields are
314/// required — `Gaussian` needs `center`/`width`/`peak-concentration`,
315/// `Dirac` needs `time`/`amount`, `None` needs nothing else.
316fn injection_to_value(injection: &InjectionBuilder, context: &str) -> anyhow::Result<Value> {
317    let injection_type = require(injection.injection_type.clone(), "type", context)?;
318
319    let mut m = Mapping::new();
320    m.insert("type".into(), injection_type.clone().into());
321
322    match injection_type.as_str() {
323        "Gaussian" => {
324            m.insert(
325                "center".into(),
326                require(injection.center, "center", context)?.into(),
327            );
328            m.insert(
329                "width".into(),
330                require(injection.width, "width", context)?.into(),
331            );
332            m.insert(
333                "peak_concentration".into(),
334                require(injection.peak_concentration, "peak-concentration", context)?.into(),
335            );
336        }
337        "Dirac" => {
338            m.insert(
339                "time".into(),
340                require(injection.time, "time", context)?.into(),
341            );
342            m.insert(
343                "amount".into(),
344                require(injection.amount, "amount", context)?.into(),
345            );
346        }
347        "None" => {}
348        other => {
349            return Err(anyhow!(
350                "{context}: unknown injection type '{other}' (expected Gaussian, Dirac, or None)"
351            ));
352        }
353    }
354
355    Ok(Value::Mapping(m))
356}

376    fn model_occurrence(discriminant: &str, pairs: &[(&str, &str)]) -> OptionOccurrence {
377        OptionOccurrence {
378            discriminant: discriminant.to_string(),
379            params: pairs
380                .iter()
381                .map(|(k, v)| (k.to_string(), v.to_string()))
382                .collect(),
383        }
384    }
385
386    fn model_args(occurrences: Vec<OptionOccurrence>) -> ParsedArgs {
387        let mut map = HashMap::new();
388        map.insert("model".to_string(), ParsedValue::Repeated(occurrences));
389        ParsedArgs::new(map)
390    }
391
392    fn save_args(project_dir: &std::path::Path, target: &str, file: &str) -> ParsedArgs {
393        ParsedArgs::from_scalars(HashMap::from([
394            (
395                "project-dir".to_string(),
396                project_dir.to_string_lossy().into_owned(),
397            ),
398            ("target".to_string(), target.to_string()),
399            ("file".to_string(), file.to_string()),
400        ]))
401    }
