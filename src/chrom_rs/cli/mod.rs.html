1//! Command-line interface for `chrom-rs`.
2//!
3//! Assembles the `dynamic-cli` application from a declarative YAML
4//! configuration embedded at compile time and wires it to the simulation
5//! pipeline defined in the [`run`](crate::cli::run) module.
6//!
7//! # Module layout (since the v0.6.0 consolidation)
8//!
9//! One file per command handler, plus two shared-concern modules:
10//!
11//! - [`context`]: [`ChromContext`](context::ChromContext) — the execution
12//!   context every handler downcasts to, including the three
13//!   pending-configuration builder slots (DD-016).
14//! - [`support`]: helpers used by more than one handler
15//!   (`to_cli_err`/`resolve_source_optional`/`path_to_str`).
16//! - [`builders`]: the pending-configuration builder *types* themselves
17//!   (`ModelBuilder`, `SolverBuilder`, `ScenarioBuilder`, ...).
18//! - [`run`], [`check`], [`config`], [`save`]: one command each.
19//!
20//! # Entry point
21//!
22//! ```rust,no_run
23//! chrom_rs::cli::build_app()
24//!     .expect("CLI initialisation failed")
25//!     .run();
26//! ```
27//!
28//! # Command surface (v0.6.0, in progress)
29//!
30//! ```text
31//! chrom-rs run [--project-dir <dir>]
32//!              --model    <file.yml>              (or --source model    file=<file.yml>)
33//!              --scenario <file.yml>              (or --source scenario file=<file.yml>)
34//!              --solver   <file.yml>              (or --source solver   file=<file.yml>)
35//!              [--output-csv   <file.csv>]        (or --output csv  file=<file.csv>)
36//!              [--output-plot  <file.png|.svg>]   (or --output png  file=<file.png>,
37//!                                                      --output svg  file=<file.svg>)
38//!              [--export-json  <file.json>]       (or --output json file=<file.json>)
39//!
40//! chrom-rs check [--project-dir <dir>]
41//!                [--source model    file=<file.yml>]
42//!                [--source scenario file=<file.yml>]
43//!                [--source solver   file=<file.yml>]
44//!
45//! chrom-rs config (alias build)
46//!                  [--model single lambda=... langmuir-k=... port-number=...
47//!                                  column-length=... n-points=... dz=... fe=... ue=...]
48//!                  [--model multi n-points=... porosity=... velocity=...
49//!                                 column-length=... dz=... fe=... ue=... stationary-fraction=...]
50//!                  [--model species name=... lambda=... langmuir-k=... port-number=...]
51//!                  [--solver RK4 total-time=... time-steps=... [step=...]]
52//!                  [--solver Euler total-time=... time-steps=... [step=...]]
53//!                  [--initial-condition zero]
54//!                  [--injection default type=... center=... width=... peak-concentration=... time=... amount=...]
55//!                  [--injection species-override species=... type=... ...]
56//!
57//! chrom-rs save --target <model-single|model-multi|solver|scenario>
58//!               --file <file.yml> [--project-dir <dir>]
59//! ```
60//!
61//! `run` accepts either the legacy scalar options or the repeatable
62//! `--source`/`--output` syntax for each role — never both for the same
63//! role in the same invocation. `check` only has the repeatable syntax,
64//! and every role is optional: with none given it just lists the project
65//! directory's config-like files; with any given, it validates exactly
66//! those. Both draw on `dynamic-cli` 0.6.0's repeatable-option-with-
67//! sub-parameters feature (see [dcli#21](https://github.com/biface/dcli/issues/21)
68//! for that feature's own design rationale on the `dynamic-cli` side).
69//!
70//! `config` (DD-016, [#53](https://github.com/biface/chromatography/issues/53))
71//! accumulates unvalidated configuration state across one or more
72//! occurrences — in a single invocation or across a chained sequence
73//! (`dynamic-cli` 0.9.0 command chaining, DD-026) — into
74//! [`ChromContext`](crate::cli::context::ChromContext)'s pending slots.
75//! `--model single`/`multi`/`species`, `--solver`, and
76//! `--initial-condition`/`--injection` are all wired up
77//! ([#68](https://github.com/biface/chromatography/issues/68),
78//! [#69](https://github.com/biface/chromatography/issues/69),
79//! [#70](https://github.com/biface/chromatography/issues/70),
80//! [#71](https://github.com/biface/chromatography/issues/71)). `save`
81//! ([#72](https://github.com/biface/chromatography/issues/72)) serialises
82//! any of those pending slots to a real file, validating required fields
83//! and erroring by name rather than writing a YAML `null`. `config` and
84//! `save` are typically chained in one invocation (DD-026): the pending
85//! state built by one or more `config` calls only needs to survive until
86//! the matching `save` call later in the same chain.
87

94/// The `run` command handler ([`RunHandler`](crate::cli::run::RunHandler)) —

96pub mod run;

98/// The `check` command handler ([`CheckHandler`](crate::cli::check::CheckHandler)) —
99/// validates configuration files without running a simulation.
100pub mod check;

102/// Pending-configuration builder types for the interactive `config`/`build`
103/// command — accumulated, unvalidated state held in
104/// [`ChromContext`](crate::cli::context::ChromContext) across chained
105/// invocations. See DD-016 (issue #53) and issue #67.
106pub mod builders;
107
108/// The `config`/`build` command handler
109/// ([`ConfigHandler`](crate::cli::config::ConfigHandler)) — builds
110/// model/solver/scenario configuration interactively. See issue #68 (and
111/// #69–72 for the parts not wired up yet).
112pub mod config;
113
114/// The `save` command handler
115/// ([`SaveHandler`](crate::cli::save::SaveHandler)) — serialises a
116/// pending `config`/`build` builder slot to a real `model.yml`/
117/// `solver.yml`/`scenario.yml` file. See issue #72.
118pub mod save;

130// ============================================================================
131// Embedded command configuration
132// ============================================================================
133
134/// YAML command configuration, embedded at compile time from
135/// `src/cli/commands.yml`.
136///
137/// Parsed once in [`build_app`] via `load_yaml`. Keeping the declarations in
138/// YAML lets maintainers adjust help text, aliases, and option metadata
139/// without touching Rust code.
140const COMMANDS_YML: &str = include_str!("commands.yml");
141
142/// Handler name that must match the `implementation:` field of the `run`
143/// command in `commands.yml`.
144const RUN_HANDLER_NAME: &str = "run_handler";
145
146/// Handler name that must match the `implementation:` field of the `check`
147/// command in `commands.yml`.
148const CHECK_HANDLER_NAME: &str = "check_handler";
149
150/// Handler name that must match the `implementation:` field of the `config`
151/// command in `commands.yml`.
152const CONFIG_HANDLER_NAME: &str = "config_handler";
153
154/// Handler name that must match the `implementation:` field of the `save`
155/// command in `commands.yml`.
156const SAVE_HANDLER_NAME: &str = "save_handler";
157
158// ============================================================================
159// build_app
160// ============================================================================
161
162/// Assembles and returns the fully configured [`CliApp`].
163///
164/// Parses the embedded command YAML, wires
165/// [`RunHandler`], [`CheckHandler`], and a fresh
166/// [`ChromContext`], then delegates to
167/// `CliBuilder::build`.
168///
169/// # Errors
170///
171/// - The embedded YAML is malformed (compile-time regression).
172/// - The builder detects a missing required handler.
173pub fn build_app() -> anyhow::Result<CliApp> {
174    let config =
175        load_yaml(COMMANDS_YML).map_err(|e| anyhow!("embedded commands.yml is invalid: {e}"))?;
176
177    CliBuilder::new()
178        .config(config)
179        .context(Box::new(ChromContext::new()))
180        .register_sync_handler(RUN_HANDLER_NAME, Box::new(RunHandler))
181        .register_sync_handler(CHECK_HANDLER_NAME, Box::new(CheckHandler))
182        .register_sync_handler(CONFIG_HANDLER_NAME, Box::new(ConfigHandler))
183        .register_sync_handler(SAVE_HANDLER_NAME, Box::new(SaveHandler))
184        .build()
185        .map_err(|e| anyhow!("CLI builder error: {e}"))
186}

192#[cfg(test)]
193mod tests {
194    use super::*;
195
196    #[test]
197    fn test_build_app_succeeds() {
198        assert!(build_app().is_ok());
199    }
200
201    #[test]
202    fn test_commands_yml_is_valid_yaml() {
203        use dynamic_cli::config::loader::load_yaml;
204        let config = load_yaml(COMMANDS_YML).expect("COMMANDS_YML must be valid");
205        assert!(config.commands.iter().any(|c| c.name == "run"));
206        assert!(config.commands.iter().any(|c| c.name == "check"));
207        assert!(config.commands.iter().any(|c| c.name == "config"));
208        assert!(config.commands.iter().any(|c| c.name == "save"));
209    }
