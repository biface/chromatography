1//! Surcoût structurel de `LangmuirMulti` face à `LangmuirSingle` à
2//! `n_species=1` (issue #55, point 1 — "est-ce utile d'avoir deux
3//! structures ?").
4//! *Structural overhead of `LangmuirMulti` vs `LangmuirSingle` at
5//! `n_species=1`.*
6//!
7//! Run with `cargo bench --bench langmuir_performance -- bench_single_vs_multi_1species`
8//! then `cargo run --bin plot_single_vs_multi --release`; output lands in
9//! `target/plots/single_vs_multi.svg`.
10//!
11//! Reads the four fixed Criterion result directories directly — this bench
12//! group has no swept dimension (unlike `bench_parallelism_threshold` or
13//! `bench_multi_species_scaling`), so there is nothing to regress and no
14//! `npts_X`-style directory parsing needed.
15//!
16//! Deliberately does **not** cross-reference `bench_multi_species_scaling`'s
17//! own `n_species=1` point: that group fixes a different `n_points` than
18//! this one (confirmed by inspecting both `langmuir_performance.rs` bench
19//! functions — not assumed), so the two are not directly comparable in
20//! absolute terms. Mixing them on one chart would silently imply a
21//! comparison that isn't valid.
22//!
23//! # Visual elements
24//!
25//! 1. **Grouped bars**: Single (blue) vs Multi (red) mean time, one group
26//!    per solver (Euler, RK4)
27//! 2. **Surcoût annotation**: Multi/Single ratio printed above each group
28//!
29//! # Cargo.toml
30//!
31//! ```toml
32//! [[bin]]
33//! name = "plot_single_vs_multi"
34//! path = "tools/plot_single_vs_multi.rs"
35//!
36//! [dependencies]
37//! plotters   = "0.3"
38//! serde      = { version = "1", features = ["derive"] }
39//! serde_json = "1"
40//! anyhow     = "1"
41//! ```

50// Désérialisation JSON Criterion / Criterion JSON deserialisation

54#[derive(Debug, Deserialize)]
55struct ConfidenceInterval {
56    lower_bound: f64,
57    upper_bound: f64,
58}
59
60#[derive(Debug, Deserialize)]
61struct Estimate {
62    confidence_interval: ConfidenceInterval,
63    point_estimate: f64,
64}
65
66#[derive(Debug, Deserialize)]
67struct Estimates {
68    mean: Estimate,
69}
70
71fn read_estimates(path: &Path) -> anyhow::Result<Estimates> {
72    let content = fs::read_to_string(path).map_err(|e| {
73        anyhow::anyhow!(
74            "cannot read {}: {e}\nRun first: cargo bench --bench langmuir_performance -- \
75             bench_single_vs_multi_1species",
76            path.display()
77        )
78    })?;
79    Ok(serde_json::from_str(&content)?)
80}

86struct SolverGroup {
87    solver: &'static str,
88    single_us: f64,
89    single_ci_us: (f64, f64),
90    multi_us: f64,
91    multi_ci_us: (f64, f64),
92}
93
94const GROUP: &str = "bench_single_vs_multi_1species";
95
96/// Lit les 4 chemins fixes (pas de balayage dans ce groupe) et retourne les
97/// deux couples Single/Multi, un par solveur.
98/// *Reads the 4 fixed paths (no sweep in this group) and returns the two
99/// Single/Multi pairs, one per solver.*
100fn collect_data(criterion_dir: &Path) -> anyhow::Result<Vec<SolverGroup>> {
101    let mut groups = Vec::new();
102
103    for (solver, single_fn, multi_fn) in [
104        ("euler", "single_euler", "multi_1sp_euler"),
105        ("rk4", "single_rk4", "multi_1sp_rk4"),
106    ] {
107        let single_path = criterion_dir
108            .join(GROUP)
109            .join(single_fn)
110            .join("new")
111            .join("estimates.json");
112        let multi_path = criterion_dir
113            .join(GROUP)
114            .join(multi_fn)
115            .join("new")
116            .join("estimates.json");
117
118        let single = read_estimates(&single_path)?;
119        let multi = read_estimates(&multi_path)?;
120
121        groups.push(SolverGroup {
122            solver,
123            single_us: single.mean.point_estimate / 1e3,
124            single_ci_us: (
125                single.mean.confidence_interval.lower_bound / 1e3,
126                single.mean.confidence_interval.upper_bound / 1e3,
127            ),
128            multi_us: multi.mean.point_estimate / 1e3,
129            multi_ci_us: (
130                multi.mean.confidence_interval.lower_bound / 1e3,
131                multi.mean.confidence_interval.upper_bound / 1e3,
132            ),
133        });
134    }
135
136    Ok(groups)
137}

143fn generate_plot(groups: &[SolverGroup], output_path: &Path) -> anyhow::Result<()> {
144    if let Some(parent) = output_path.parent() {
145        fs::create_dir_all(parent)?;
146    }
147
148    let y_max = groups
149        .iter()
150        .map(|g| g.single_us.max(g.multi_us))
151        .fold(0.0_f64, f64::max)
152        * 1.35;
153
154    let root = SVGBackend::new(output_path, (900, 650)).into_drawing_area();
155    root.fill(&WHITE)?;
156
157    let mut chart = ChartBuilder::on(&root)
158        .margin(50)
159        .x_label_area_size(45)
160        .y_label_area_size(80)
161        .caption(
162            "LangmuirSingle vs LangmuirMulti (n_species=1) — surcoût structurel",
163            ("sans-serif", 18).into_font(),
164        )
165        .build_cartesian_2d(0f64..groups.len() as f64, 0f64..y_max)?;
166
167    chart
168        .configure_mesh()
169        .disable_x_mesh()
170        .x_desc("")
171        .y_desc("Temps moyen (µs)")
172        .x_label_formatter(&|_| String::new())
173        .y_label_formatter(&|v| format!("{v:.0} µs"))
174        .draw()?;
175
176    let bar_half_width = 0.35;
177
178    for (i, g) in groups.iter().enumerate() {
179        let x0 = i as f64;
180        let single_center = x0 + 0.5 - 0.22;
181        let multi_center = x0 + 0.5 + 0.22;
182
183        chart.draw_series(std::iter::once(Rectangle::new(
184            [
185                (single_center - bar_half_width / 2.0, 0.0),
186                (single_center + bar_half_width / 2.0, g.single_us),
187            ],
188            BLUE.mix(0.7).filled(),
189        )))?;
190        chart.draw_series(std::iter::once(Rectangle::new(
191            [
192                (multi_center - bar_half_width / 2.0, 0.0),
193                (multi_center + bar_half_width / 2.0, g.multi_us),
194            ],
195            RED.mix(0.7).filled(),
196        )))?;
197
198        // Barres d'IC 95% / 95% CI whiskers
199        chart.draw_series(std::iter::once(PathElement::new(
200            vec![
201                (single_center, g.single_ci_us.0),
202                (single_center, g.single_ci_us.1),
203            ],
204            BLACK.stroke_width(1),
205        )))?;
206        chart.draw_series(std::iter::once(PathElement::new(
207            vec![
208                (multi_center, g.multi_ci_us.0),
209                (multi_center, g.multi_ci_us.1),
210            ],
211            BLACK.stroke_width(1),
212        )))?;
213
214        chart.draw_series(std::iter::once(Text::new(
215            g.solver.to_uppercase(),
216            (x0 + 0.5, -y_max * 0.04),
217            ("sans-serif", 13).into_font().color(&BLACK),
218        )))?;
219
220        let ratio = g.multi_us / g.single_us;
221        let top = g.single_us.max(g.multi_us);
222        chart.draw_series(std::iter::once(Text::new(
223            format!("Surcoût ×{ratio:.2}"),
224            (x0 + 0.5, top + y_max * 0.05),
225            ("sans-serif", 13).into_font().color(&RGBColor(150, 0, 0)),
226        )))?;
227    }
228
229    // Légende manuelle, même convention que plot_cost_accuracy.rs
230    // Manual legend, same convention as plot_cost_accuracy.rs
231    let w = groups.len() as f64;
232    chart.draw_series(std::iter::once(Rectangle::new(
233        [(0.02 * w, y_max * 0.94), (0.06 * w, y_max * 0.98)],
234        BLUE.mix(0.7).filled(),
235    )))?;
236    chart.draw_series(std::iter::once(Text::new(
237        "Single",
238        (0.08 * w, y_max * 0.96),
239        ("sans-serif", 12).into_font(),
240    )))?;
241    chart.draw_series(std::iter::once(Rectangle::new(
242        [(0.22 * w, y_max * 0.94), (0.26 * w, y_max * 0.98)],
243        RED.mix(0.7).filled(),
244    )))?;
245    chart.draw_series(std::iter::once(Text::new(
246        "Multi (1 espèce)",
247        (0.28 * w, y_max * 0.96),
248        ("sans-serif", 12).into_font(),
249    )))?;
250
251    root.present()?;
252    println!("Plot generated: {}", output_path.display());
253    Ok(())
254}

256// =================================================================================================
257// Entry point
258// =================================================================================================
259
260fn main() -> anyhow::Result<()> {
261    // Standalone binary variant (Variant B) — see
262    // `chrom_rs::output::visualization::fonts` for why this call is needed
263    // before any chart renders.
264    chrom_rs::output::register_fonts();
265
266    let criterion_dir = PathBuf::from("target/criterion");
267    println!("Reading Criterion data from {}...", criterion_dir.display());
268
269    let groups = collect_data(&criterion_dir)?;
270
271    println!(
272        "\n{:<8} {:>14} {:>14} {:>10}",
273        "solver", "single (µs)", "multi (µs)", "ratio"
274    );
275    println!("{:-<50}", "");
276    for g in &groups {
277        println!(
278            "{:<8} {:>14.2} {:>14.2} {:>9.2}×",
279            g.solver,
280            g.single_us,
281            g.multi_us,
282            g.multi_us / g.single_us
283        );
284    }
285
286    let output_path = PathBuf::from("target/plots/single_vs_multi.svg");
287    println!("\nGenerating plot...");
288    generate_plot(&groups, &output_path)?;
289    Ok(())
290}

296#[cfg(test)]
297mod tests {
298    /// Régression sur les données réelles (V6, 28/08) : le surcoût structurel
299    /// de Multi à n_species=1 doit rester dans une fourchette large mais
300    /// non triviale — s'il tombe sous ×2, la distinction Single/Multi perd
301    /// une partie de sa justification (issue #55, point 1) ; s'il explose
302    /// au-delà de ×20, quelque chose a régressé ailleurs (allocation,
303    /// dispatch...).
304    /// *Regression on real data (V6, 28/08): Multi's structural overhead at
305    /// n_species=1 should stay within a wide but non-trivial range — below
306    /// ×2 the Single/Multi distinction loses part of its justification
307    /// (issue #55, point 1); above ×20 something regressed elsewhere
308    /// (allocation, dispatch...).*
309    #[test]
310    fn test_regression_overhead_from_actual_data() {
311        let single_euler_ns = 861_371.25_f64;
312        let multi_euler_ns = 6_864_477.56_f64;
313        let ratio_euler = multi_euler_ns / single_euler_ns;
314        assert!(
315            (2.0..20.0).contains(&ratio_euler),
316            "surcoût Euler hors fourchette plausible: {ratio_euler:.2}×"
317        );
318
319        let single_rk4_ns = 2_786_793.48_f64;
320        let multi_rk4_ns = 26_751_994.53_f64;
321        let ratio_rk4 = multi_rk4_ns / single_rk4_ns;
322        assert!(
323            (2.0..20.0).contains(&ratio_rk4),
324            "surcoût RK4 hors fourchette plausible: {ratio_rk4:.2}×"
325        );
326    }
327
328    /// Conversion ns -> µs cohérente (pas de coquille /1e6 copiée depuis les
329    /// autres outils, qui convertissent en ms).
330    /// *Consistent ns -> µs conversion (no /1e6 typo copied from the other
331    /// tools, which convert to ms).*
332    #[test]
333    fn test_ns_to_us() {
334        assert!((1_000.0_f64 / 1e3 - 1.0).abs() < 1e-12);
335    }
